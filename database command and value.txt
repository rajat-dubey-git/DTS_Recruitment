/* =============================================================================
   DTS Recruitment & Staffing Platform
   ========================================================================== */

SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- DROP DATABASE IF EXISTS dts_recruitment;
CREATE DATABASE dts_recruitment
  DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE dts_recruitment;

/* ============================ 1. USERS & CUSTOMERS ======================== */

CREATE TABLE users (
  user_id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  full_name      VARCHAR(100) NOT NULL,
  email          VARCHAR(150) NOT NULL,
  mobile         VARCHAR(15)  NULL,
  password_hash  VARCHAR(255) NULL,
  role           ENUM('ADMIN','RECRUITER','CANDIDATE','CUSTOMER') NOT NULL,
  status         ENUM('INVITED','ACTIVE','INACTIVE','LOCKED') NOT NULL DEFAULT 'INVITED',
  created_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  last_login_at  DATETIME NULL,
  updated_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT pk_users PRIMARY KEY (user_id),
  CONSTRAINT uq_users_email UNIQUE (email),
  KEY ix_users_role_status (role, status),
  CONSTRAINT ck_users_email  CHECK (email LIKE '%_@_%._%'),
  CONSTRAINT ck_users_mobile CHECK (mobile IS NULL OR REGEXP_LIKE(mobile, '^[+]?[0-9]{7,15}$'))
) ENGINE=InnoDB;

CREATE TABLE customers (
  customer_id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  customer_code      VARCHAR(30)  NOT NULL,
  company_name       VARCHAR(200) NOT NULL,
  gstin              VARCHAR(15)  NULL,
  pan                VARCHAR(10)  NULL,
  industry           VARCHAR(100) NULL,
  payment_terms_days SMALLINT UNSIGNED NOT NULL DEFAULT 30,
  status             ENUM('PROSPECT','ACTIVE','INACTIVE') NOT NULL DEFAULT 'PROSPECT',
  account_owner_id   INT UNSIGNED NULL,
  created_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT pk_customers PRIMARY KEY (customer_id),
  CONSTRAINT uq_customers_code  UNIQUE (customer_code),
  CONSTRAINT uq_customers_gstin UNIQUE (gstin),
  KEY ix_customers_status (status),
  CONSTRAINT fk_customers_owner FOREIGN KEY (account_owner_id) REFERENCES users (user_id),
  CONSTRAINT ck_customers_terms CHECK (payment_terms_days <= 365),
  CONSTRAINT ck_customers_gstin CHECK (gstin IS NULL OR REGEXP_LIKE(gstin, '^[0-9]{2}[A-Z]{5}[0-9]{4}[A-Z][1-9A-Z]Z[0-9A-Z]$', 'c')),
  CONSTRAINT ck_customers_pan   CHECK (pan   IS NULL OR REGEXP_LIKE(pan,   '^[A-Z]{5}[0-9]{4}[A-Z]$', 'c')),
  /* a GSTIN embeds the PAN at positions 3-12; keep the two consistent */
  CONSTRAINT ck_customers_gstin_pan CHECK (gstin IS NULL OR pan IS NULL OR SUBSTRING(gstin, 3, 10) = pan)
) ENGINE=InnoDB;

CREATE TABLE customer_addresses (
  address_id    INT UNSIGNED NOT NULL AUTO_INCREMENT,
  customer_id   INT UNSIGNED NOT NULL,
  address_type  ENUM('REGISTERED','BILLING','SITE','OTHER') NOT NULL,
  address_line  VARCHAR(255) NOT NULL,
  city          VARCHAR(100) NOT NULL,
  state         VARCHAR(100) NULL,
  postal_code   VARCHAR(10)  NULL,
  is_primary    TINYINT UNSIGNED NOT NULL DEFAULT 0,
  primary_marker TINYINT GENERATED ALWAYS AS (IF(is_primary = 1, 1, NULL)) VIRTUAL,
  CONSTRAINT pk_customer_addresses PRIMARY KEY (address_id),
  CONSTRAINT uq_customer_addresses_primary UNIQUE (customer_id, address_type, primary_marker),
  CONSTRAINT fk_customer_addresses_customer FOREIGN KEY (customer_id) REFERENCES customers (customer_id),
  CONSTRAINT ck_customer_addresses_primary CHECK (is_primary IN (0,1))
) ENGINE=InnoDB;

CREATE TABLE customer_contacts (
  contact_id    INT UNSIGNED NOT NULL AUTO_INCREMENT,
  customer_id   INT UNSIGNED NOT NULL,
  user_id       INT UNSIGNED NOT NULL,
  designation   VARCHAR(100) NULL,
  contact_type  ENUM('HIRING','BILLING','SIGNATORY','TECHNICAL','OTHER') NOT NULL DEFAULT 'HIRING',
  is_primary    TINYINT UNSIGNED NOT NULL DEFAULT 0,
  status        ENUM('ACTIVE','INACTIVE') NOT NULL DEFAULT 'ACTIVE',
  primary_marker TINYINT GENERATED ALWAYS AS (IF(is_primary = 1, 1, NULL)) VIRTUAL,
  CONSTRAINT pk_customer_contacts PRIMARY KEY (contact_id),
  CONSTRAINT uq_customer_contacts_user UNIQUE (user_id),
  CONSTRAINT uq_customer_contacts_primary UNIQUE (customer_id, primary_marker),
  CONSTRAINT fk_customer_contacts_customer FOREIGN KEY (customer_id) REFERENCES customers (customer_id),
  CONSTRAINT fk_customer_contacts_user     FOREIGN KEY (user_id)     REFERENCES users (user_id),
  CONSTRAINT ck_customer_contacts_primary CHECK (is_primary IN (0,1))
) ENGINE=InnoDB;

/* ====================== 2. MASTER DATA (admin set-up) ===================== */

CREATE TABLE skills (
  skill_id       INT UNSIGNED NOT NULL AUTO_INCREMENT,
  skill_name     VARCHAR(100) NOT NULL,
  skill_category VARCHAR(60)  NULL,
  CONSTRAINT pk_skills PRIMARY KEY (skill_id),
  CONSTRAINT uq_skills_name UNIQUE (skill_name)
) ENGINE=InnoDB;

CREATE TABLE service_types (
  service_type_id  INT UNSIGNED NOT NULL AUTO_INCREMENT,
  service_code     VARCHAR(30)  NOT NULL,
  service_name     VARCHAR(100) NOT NULL,
  employment_model ENUM('DTS_PAYROLL','CUSTOMER_PAYROLL') NOT NULL,
  CONSTRAINT pk_service_types PRIMARY KEY (service_type_id),
  CONSTRAINT uq_service_types_code UNIQUE (service_code)
) ENGINE=InnoDB;

CREATE TABLE job_profiles (
  job_profile_id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  profile_code          VARCHAR(30)  NOT NULL,
  profile_name          VARCHAR(150) NOT NULL,
  category              VARCHAR(100) NULL,
  purpose               TEXT NOT NULL,
  key_responsibilities  TEXT NOT NULL,
  qualifications        TEXT NULL,
  reporting_to          VARCHAR(150) NULL,
  default_experience_min DECIMAL(4,1) NULL,
  default_experience_max DECIMAL(4,1) NULL,
  status                ENUM('DRAFT','ACTIVE','ARCHIVED') NOT NULL DEFAULT 'DRAFT',
  created_by            INT UNSIGNED NULL,
  created_at            DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT pk_job_profiles PRIMARY KEY (job_profile_id),
  CONSTRAINT uq_job_profiles_code UNIQUE (profile_code),
  KEY ix_job_profiles_status (status),
  CONSTRAINT fk_job_profiles_created_by FOREIGN KEY (created_by) REFERENCES users (user_id),
  CONSTRAINT ck_job_profiles_exp CHECK (default_experience_min IS NULL OR default_experience_max IS NULL
                                        OR default_experience_max >= default_experience_min),
  CONSTRAINT ck_job_profiles_exp_pos CHECK (default_experience_min IS NULL OR default_experience_min >= 0)
) ENGINE=InnoDB;

CREATE TABLE job_profile_skills (
  job_profile_id       INT UNSIGNED NOT NULL,
  skill_id             INT UNSIGNED NOT NULL,
  is_mandatory         TINYINT UNSIGNED NOT NULL DEFAULT 1,
  min_experience_years DECIMAL(4,1) NULL,
  CONSTRAINT pk_job_profile_skills PRIMARY KEY (job_profile_id, skill_id),
  CONSTRAINT fk_job_profile_skills_profile FOREIGN KEY (job_profile_id) REFERENCES job_profiles (job_profile_id) ON DELETE CASCADE,
  CONSTRAINT fk_job_profile_skills_skill   FOREIGN KEY (skill_id)       REFERENCES skills (skill_id),
  CONSTRAINT ck_job_profile_skills_mand CHECK (is_mandatory IN (0,1)),
  CONSTRAINT ck_job_profile_skills_exp  CHECK (min_experience_years IS NULL OR min_experience_years >= 0)
) ENGINE=InnoDB;

/* ================== 3. AGREEMENT, PRICING, SERVICE ORDER ================== */

CREATE TABLE agreements (
  agreement_id     INT UNSIGNED NOT NULL AUTO_INCREMENT,
  agreement_number VARCHAR(50)  NOT NULL,
  customer_id      INT UNSIGNED NOT NULL,
  title            VARCHAR(200) NOT NULL,
  status           ENUM('DRAFT','PENDING_SIGNATURE','ACTIVE','EXPIRED','TERMINATED') NOT NULL DEFAULT 'DRAFT',
  effective_date   DATE NULL,
  expiry_date      DATE NULL,
  signed_date      DATE NULL,
  auto_renew       TINYINT UNSIGNED NOT NULL DEFAULT 0,
  document_path    VARCHAR(500) NULL,
  created_by       INT UNSIGNED NULL,
  created_at       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT pk_agreements PRIMARY KEY (agreement_id),
  CONSTRAINT uq_agreements_number UNIQUE (agreement_number),
  KEY ix_agreements_customer_status (customer_id, status),
  CONSTRAINT fk_agreements_customer   FOREIGN KEY (customer_id) REFERENCES customers (customer_id),
  CONSTRAINT fk_agreements_created_by FOREIGN KEY (created_by)  REFERENCES users (user_id),
  CONSTRAINT ck_agreements_dates  CHECK (expiry_date IS NULL OR effective_date IS NULL OR expiry_date >= effective_date),
  CONSTRAINT ck_agreements_active CHECK (status <> 'ACTIVE' OR (effective_date IS NOT NULL AND signed_date IS NOT NULL)),
  CONSTRAINT ck_agreements_renew  CHECK (auto_renew IN (0,1))
) ENGINE=InnoDB;

CREATE TABLE rate_cards (
  rate_card_id      INT UNSIGNED NOT NULL AUTO_INCREMENT,
  agreement_id      INT UNSIGNED NOT NULL,
  service_type_id   INT UNSIGNED NOT NULL,
  job_profile_id    INT UNSIGNED NOT NULL,
  fee_type          ENUM('PERCENTAGE','FIXED_AMOUNT','SALARY_MONTHS') NOT NULL,
  fee_basis         ENUM('MONTHLY_SALARY','ANNUAL_CTC','HOURLY','PER_PLACEMENT') NOT NULL,
  fee_value         DECIMAL(12,2) NOT NULL,
  billing_frequency ENUM('ONE_TIME','MONTHLY','QUARTERLY') NOT NULL,
  replacement_days  SMALLINT UNSIGNED NULL,
  effective_from    DATE NOT NULL,
  effective_to      DATE NULL,
  CONSTRAINT pk_rate_cards PRIMARY KEY (rate_card_id),
  CONSTRAINT uq_rate_cards_key UNIQUE (agreement_id, service_type_id, job_profile_id, effective_from),
  KEY ix_rate_cards_service (service_type_id),
  KEY ix_rate_cards_profile (job_profile_id),
  CONSTRAINT fk_rate_cards_agreement FOREIGN KEY (agreement_id)    REFERENCES agreements (agreement_id),
  CONSTRAINT fk_rate_cards_service   FOREIGN KEY (service_type_id) REFERENCES service_types (service_type_id),
  CONSTRAINT fk_rate_cards_profile   FOREIGN KEY (job_profile_id)  REFERENCES job_profiles (job_profile_id),
  CONSTRAINT ck_rate_cards_value CHECK (fee_value >= 0 AND (fee_type <> 'PERCENTAGE' OR fee_value <= 100)),
  CONSTRAINT ck_rate_cards_dates CHECK (effective_to IS NULL OR effective_to >= effective_from)
) ENGINE=InnoDB;

CREATE TABLE service_orders (
  service_order_id   INT UNSIGNED NOT NULL AUTO_INCREMENT,
  order_number       VARCHAR(50)  NOT NULL,
  agreement_id       INT UNSIGNED NOT NULL,
  service_type_id    INT UNSIGNED NOT NULL,
  customer_reference VARCHAR(100) NULL,
  title              VARCHAR(200) NOT NULL,
  scope_description  TEXT NULL,
  start_date         DATE NULL,
  end_date           DATE NULL,
  status             ENUM('DRAFT','PENDING_APPROVAL','APPROVED','IN_PROGRESS','COMPLETED','CANCELLED') NOT NULL DEFAULT 'DRAFT',
  approved_by        INT UNSIGNED NULL,
  approved_at        DATETIME NULL,
  created_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT pk_service_orders PRIMARY KEY (service_order_id),
  CONSTRAINT uq_service_orders_number UNIQUE (order_number),
  KEY ix_service_orders_agreement (agreement_id, status),
  KEY ix_service_orders_service (service_type_id),
  CONSTRAINT fk_service_orders_agreement   FOREIGN KEY (agreement_id)    REFERENCES agreements (agreement_id),
  CONSTRAINT fk_service_orders_service     FOREIGN KEY (service_type_id) REFERENCES service_types (service_type_id),
  CONSTRAINT fk_service_orders_approved_by FOREIGN KEY (approved_by)     REFERENCES users (user_id),
  CONSTRAINT ck_service_orders_dates    CHECK (end_date IS NULL OR start_date IS NULL OR end_date >= start_date),
  CONSTRAINT ck_service_orders_approval CHECK ((approved_by IS NULL) = (approved_at IS NULL))
) ENGINE=InnoDB;

/* ========================= 4. JOB REQUIREMENTS ============================ */

CREATE TABLE job_requirements (
  requirement_id       INT UNSIGNED NOT NULL AUTO_INCREMENT,
  requirement_number   VARCHAR(50)  NOT NULL,
  service_order_id     INT UNSIGNED NOT NULL,
  job_profile_id       INT UNSIGNED NOT NULL,
  raised_by_user_id    INT UNSIGNED NULL,
  job_title            VARCHAR(150) NOT NULL,
  job_description      TEXT NOT NULL,
  headcount            SMALLINT UNSIGNED NOT NULL DEFAULT 1,
  work_mode            ENUM('ONSITE','REMOTE','HYBRID') NOT NULL,
  work_location        VARCHAR(150) NULL,
  experience_min_years DECIMAL(4,1) NULL,
  experience_max_years DECIMAL(4,1) NULL,
  salary_min           DECIMAL(12,2) NULL,
  salary_max           DECIMAL(12,2) NULL,
  employment_schedule  ENUM('FULL_TIME','PART_TIME') NOT NULL,
  hours_per_week       DECIMAL(5,2) NULL,
  target_join_date     DATE NULL,
  status               ENUM('DRAFT','PENDING_APPROVAL','APPROVED','OPEN','ON_HOLD','FILLED','CLOSED','CANCELLED') NOT NULL DEFAULT 'DRAFT',
  approved_by          INT UNSIGNED NULL,
  approved_at          DATETIME NULL,
  published_at         DATETIME NULL,
  created_at           DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT pk_job_requirements PRIMARY KEY (requirement_id),
  CONSTRAINT uq_job_requirements_number UNIQUE (requirement_number),
  KEY ix_job_requirements_order_status (service_order_id, status),
  KEY ix_job_requirements_profile (job_profile_id),
  KEY ix_job_requirements_status_pub (status, published_at),
  CONSTRAINT fk_job_requirements_order    FOREIGN KEY (service_order_id)  REFERENCES service_orders (service_order_id),
  CONSTRAINT fk_job_requirements_profile  FOREIGN KEY (job_profile_id)    REFERENCES job_profiles (job_profile_id),
  CONSTRAINT fk_job_requirements_raised   FOREIGN KEY (raised_by_user_id) REFERENCES users (user_id),
  CONSTRAINT fk_job_requirements_approved FOREIGN KEY (approved_by)       REFERENCES users (user_id),
  CONSTRAINT ck_job_requirements_headcount CHECK (headcount > 0),
  CONSTRAINT ck_job_requirements_exp    CHECK (experience_min_years IS NULL OR experience_max_years IS NULL OR experience_max_years >= experience_min_years),
  CONSTRAINT ck_job_requirements_salary CHECK (salary_min IS NULL OR salary_max IS NULL OR salary_max >= salary_min),
  CONSTRAINT ck_job_requirements_hours  CHECK (hours_per_week IS NULL OR (hours_per_week > 0 AND hours_per_week <= 168)),
  CONSTRAINT ck_job_requirements_parttime CHECK (employment_schedule = 'FULL_TIME' OR hours_per_week IS NOT NULL),
  CONSTRAINT ck_job_requirements_approval CHECK ((approved_by IS NULL) = (approved_at IS NULL))
) ENGINE=InnoDB;

CREATE TABLE requirement_assignments (
  assignment_id   INT UNSIGNED NOT NULL AUTO_INCREMENT,
  requirement_id  INT UNSIGNED NOT NULL,
  recruiter_id    INT UNSIGNED NOT NULL,
  assignment_role ENUM('LEAD','SUPPORT') NOT NULL DEFAULT 'LEAD',
  assigned_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  unassigned_at   DATETIME NULL,
  active_marker   TINYINT GENERATED ALWAYS AS (IF(unassigned_at IS NULL, 1, NULL)) VIRTUAL,
  CONSTRAINT pk_requirement_assignments PRIMARY KEY (assignment_id),
  CONSTRAINT uq_requirement_assignments_active UNIQUE (requirement_id, recruiter_id, active_marker),
  KEY ix_requirement_assignments_recruiter (recruiter_id),
  CONSTRAINT fk_requirement_assignments_req FOREIGN KEY (requirement_id) REFERENCES job_requirements (requirement_id),
  CONSTRAINT fk_requirement_assignments_rec FOREIGN KEY (recruiter_id)   REFERENCES users (user_id),
  CONSTRAINT ck_requirement_assignments_dates CHECK (unassigned_at IS NULL OR unassigned_at >= assigned_at)
) ENGINE=InnoDB;

CREATE TABLE job_requirement_skills (
  requirement_id       INT UNSIGNED NOT NULL,
  skill_id             INT UNSIGNED NOT NULL,
  is_mandatory         TINYINT UNSIGNED NOT NULL DEFAULT 1,
  min_experience_years DECIMAL(4,1) NULL,
  CONSTRAINT pk_job_requirement_skills PRIMARY KEY (requirement_id, skill_id),
  CONSTRAINT fk_job_requirement_skills_req   FOREIGN KEY (requirement_id) REFERENCES job_requirements (requirement_id) ON DELETE CASCADE,
  CONSTRAINT fk_job_requirement_skills_skill FOREIGN KEY (skill_id)       REFERENCES skills (skill_id),
  CONSTRAINT ck_job_requirement_skills_mand CHECK (is_mandatory IN (0,1)),
  CONSTRAINT ck_job_requirement_skills_exp  CHECK (min_experience_years IS NULL OR min_experience_years >= 0)
) ENGINE=InnoDB;

/* ============================ 5. CANDIDATES =============================== */

CREATE TABLE candidates (
  candidate_id           INT UNSIGNED NOT NULL AUTO_INCREMENT,
  candidate_code         VARCHAR(30)  NOT NULL,
  first_name             VARCHAR(60)  NOT NULL,
  last_name              VARCHAR(60)  NULL,
  email                  VARCHAR(150) NULL,
  mobile                 VARCHAR(15)  NOT NULL,
  date_of_birth          DATE NULL,
  current_city           VARCHAR(100) NULL,
  current_state          VARCHAR(100) NULL,
  total_experience_years DECIMAL(4,1) NULL,
  current_ctc            DECIMAL(12,2) NULL,
  expected_ctc           DECIMAL(12,2) NULL,
  notice_period_days     SMALLINT UNSIGNED NULL,
  source                 ENUM('PORTAL','ADMIN_UPLOAD','REFERRAL','JOB_BOARD','OTHER') NOT NULL DEFAULT 'PORTAL',
  profile_summary        TEXT NULL,
  status                 ENUM('NEW','ACTIVE','PLACED','INACTIVE','BLACKLISTED') NOT NULL DEFAULT 'NEW',
  created_at             DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at             DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  created_by             INT UNSIGNED NULL,
  user_id                INT UNSIGNED NULL,
  CONSTRAINT pk_candidates PRIMARY KEY (candidate_id),
  CONSTRAINT uq_candidates_code  UNIQUE (candidate_code),
  CONSTRAINT uq_candidates_email UNIQUE (email),
  CONSTRAINT uq_candidates_user  UNIQUE (user_id),
  KEY ix_candidates_mobile (mobile),
  KEY ix_candidates_status_city (status, current_city),
  CONSTRAINT fk_candidates_created_by FOREIGN KEY (created_by) REFERENCES users (user_id),
  CONSTRAINT fk_candidates_user       FOREIGN KEY (user_id)    REFERENCES users (user_id),
  CONSTRAINT ck_candidates_mobile CHECK (REGEXP_LIKE(mobile, '^[+]?[0-9]{7,15}$')),
  CONSTRAINT ck_candidates_email  CHECK (email IS NULL OR email LIKE '%_@_%._%'),
  CONSTRAINT ck_candidates_exp    CHECK (total_experience_years IS NULL OR total_experience_years >= 0),
  CONSTRAINT ck_candidates_ctc    CHECK ((current_ctc IS NULL OR current_ctc >= 0) AND (expected_ctc IS NULL OR expected_ctc >= 0))
) ENGINE=InnoDB;

CREATE TABLE candidate_documents (
  document_id         INT UNSIGNED NOT NULL AUTO_INCREMENT,
  candidate_id        INT UNSIGNED NOT NULL,
  document_type       ENUM('RESUME','ID_PROOF','EDUCATION_CERT','EXPERIENCE_LETTER','OTHER') NOT NULL,
  file_name           VARCHAR(255) NOT NULL,
  file_path           VARCHAR(500) NOT NULL,
  is_current          TINYINT UNSIGNED NOT NULL DEFAULT 1,
  verification_status ENUM('PENDING','VERIFIED','REJECTED') NOT NULL DEFAULT 'PENDING',
  verified_by         INT UNSIGNED NULL,
  verified_at         DATETIME NULL,
  uploaded_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  uploaded_by         INT UNSIGNED NULL,
  current_resume_marker TINYINT GENERATED ALWAYS AS (IF(is_current = 1 AND document_type = 'RESUME', 1, NULL)) VIRTUAL,
  CONSTRAINT pk_candidate_documents PRIMARY KEY (document_id),
  CONSTRAINT uq_candidate_documents_resume UNIQUE (candidate_id, current_resume_marker),
  KEY ix_candidate_documents_type (candidate_id, document_type),
  CONSTRAINT fk_candidate_documents_candidate   FOREIGN KEY (candidate_id) REFERENCES candidates (candidate_id),
  CONSTRAINT fk_candidate_documents_verified_by FOREIGN KEY (verified_by)  REFERENCES users (user_id),
  CONSTRAINT fk_candidate_documents_uploaded_by FOREIGN KEY (uploaded_by)  REFERENCES users (user_id),
  CONSTRAINT ck_candidate_documents_current  CHECK (is_current IN (0,1)),
  CONSTRAINT ck_candidate_documents_verified CHECK ((verified_by IS NULL) = (verified_at IS NULL))
) ENGINE=InnoDB;

CREATE TABLE candidate_education (
  education_id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  candidate_id        INT UNSIGNED NOT NULL,
  qualification       VARCHAR(100) NOT NULL,
  specialization      VARCHAR(100) NULL,
  institution         VARCHAR(200) NULL,
  passing_year        SMALLINT UNSIGNED NULL,
  grade_or_percentage VARCHAR(30) NULL,
  CONSTRAINT pk_candidate_education PRIMARY KEY (education_id),
  KEY ix_candidate_education_candidate (candidate_id),
  CONSTRAINT fk_candidate_education_candidate FOREIGN KEY (candidate_id) REFERENCES candidates (candidate_id) ON DELETE CASCADE,
  CONSTRAINT ck_candidate_education_year CHECK (passing_year IS NULL OR passing_year BETWEEN 1950 AND 2100)
) ENGINE=InnoDB;

CREATE TABLE candidate_experience (
  experience_id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  candidate_id  INT UNSIGNED NOT NULL,
  employer_name VARCHAR(200) NOT NULL,
  designation   VARCHAR(120) NOT NULL,
  start_date    DATE NULL,
  end_date      DATE NULL,
  is_current    TINYINT UNSIGNED NOT NULL DEFAULT 0,
  CONSTRAINT pk_candidate_experience PRIMARY KEY (experience_id),
  KEY ix_candidate_experience_candidate (candidate_id),
  CONSTRAINT fk_candidate_experience_candidate FOREIGN KEY (candidate_id) REFERENCES candidates (candidate_id) ON DELETE CASCADE,
  CONSTRAINT ck_candidate_experience_dates   CHECK (end_date IS NULL OR start_date IS NULL OR end_date >= start_date),
  CONSTRAINT ck_candidate_experience_current CHECK (is_current IN (0,1) AND (is_current = 0 OR end_date IS NULL))
) ENGINE=InnoDB;

CREATE TABLE candidate_skills (
  candidate_id     INT UNSIGNED NOT NULL,
  skill_id         INT UNSIGNED NOT NULL,
  experience_years DECIMAL(4,1) NULL,
  proficiency      ENUM('BEGINNER','INTERMEDIATE','ADVANCED','EXPERT') NULL,
  CONSTRAINT pk_candidate_skills PRIMARY KEY (candidate_id, skill_id),
  KEY ix_candidate_skills_skill (skill_id),
  CONSTRAINT fk_candidate_skills_candidate FOREIGN KEY (candidate_id) REFERENCES candidates (candidate_id) ON DELETE CASCADE,
  CONSTRAINT fk_candidate_skills_skill     FOREIGN KEY (skill_id)     REFERENCES skills (skill_id),
  CONSTRAINT ck_candidate_skills_exp CHECK (experience_years IS NULL OR experience_years >= 0)
) ENGINE=InnoDB;

/* ================ 6. APPLICATION -> SCREENING -> SHORTLIST ================ */

CREATE TABLE applications (
  application_id     INT UNSIGNED NOT NULL AUTO_INCREMENT,
  application_number VARCHAR(50)  NOT NULL,
  requirement_id     INT UNSIGNED NOT NULL,
  candidate_id       INT UNSIGNED NOT NULL,
  recruiter_id       INT UNSIGNED NULL,
  status             ENUM('APPLIED','SCREENING','SHORTLISTED','INTERVIEW','SELECTED','REJECTED','WITHDRAWN') NOT NULL DEFAULT 'APPLIED',
  source             ENUM('SELF_APPLIED','RECRUITER_ADDED','REFERRAL','JOB_BOARD') NOT NULL DEFAULT 'SELF_APPLIED',
  applied_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  rejection_reason   VARCHAR(255) NULL,
  CONSTRAINT pk_applications PRIMARY KEY (application_id),
  CONSTRAINT uq_applications_number UNIQUE (application_number),
  CONSTRAINT uq_applications_req_cand UNIQUE (requirement_id, candidate_id),
  KEY ix_applications_candidate (candidate_id),
  KEY ix_applications_recruiter (recruiter_id),
  KEY ix_applications_req_status (requirement_id, status),
  CONSTRAINT fk_applications_requirement FOREIGN KEY (requirement_id) REFERENCES job_requirements (requirement_id),
  CONSTRAINT fk_applications_candidate   FOREIGN KEY (candidate_id)   REFERENCES candidates (candidate_id),
  CONSTRAINT fk_applications_recruiter   FOREIGN KEY (recruiter_id)   REFERENCES users (user_id),
  CONSTRAINT ck_applications_rejection CHECK (rejection_reason IS NULL OR status = 'REJECTED')
) ENGINE=InnoDB;

CREATE TABLE screenings (
  screening_id   INT UNSIGNED NOT NULL AUTO_INCREMENT,
  application_id INT UNSIGNED NOT NULL,
  screened_by    INT UNSIGNED NULL,

DELIMITER $$

/* a) A resume share must use a document of the SAME candidate as the application,
      and go to a contact of the SAME customer that owns the requirement.        */
CREATE PROCEDURE sp_check_resume_share(IN p_app INT UNSIGNED, IN p_doc INT UNSIGNED, IN p_contact INT UNSIGNED)
BEGIN
  DECLARE v_app_cand  INT UNSIGNED;
  DECLARE v_app_cust  INT UNSIGNED;
  DECLARE v_doc_cand  INT UNSIGNED;
  DECLARE v_cont_cust INT UNSIGNED;

  SELECT ap.candidate_id, ag.customer_id INTO v_app_cand, v_app_cust
    FROM applications ap
    JOIN job_requirements jr ON jr.requirement_id = ap.requirement_id
    JOIN service_orders so   ON so.service_order_id = jr.service_order_id
    JOIN agreements ag       ON ag.agreement_id = so.agreement_id
   WHERE ap.application_id = p_app;
  SELECT candidate_id INTO v_doc_cand  FROM candidate_documents WHERE document_id = p_doc;
  SELECT customer_id  INTO v_cont_cust FROM customer_contacts   WHERE contact_id  = p_contact;

  IF v_doc_cand <> v_app_cand THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'resume_shares: document does not belong to the application candidate';
  END IF;
  IF v_cont_cust <> v_app_cust THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'resume_shares: contact does not belong to the requirement customer';
  END IF;
END$$

CREATE TRIGGER trg_resume_shares_bi BEFORE INSERT ON resume_shares FOR EACH ROW
BEGIN CALL sp_check_resume_share(NEW.application_id, NEW.document_id, NEW.shared_with_contact_id); END$$
CREATE TRIGGER trg_resume_shares_bu BEFORE UPDATE ON resume_shares FOR EACH ROW
BEGIN CALL sp_check_resume_share(NEW.application_id, NEW.document_id, NEW.shared_with_contact_id); END$$

/* b) Invoice lines: invoice must be DRAFT, and the billed deployment/placement
      must belong to the same service order as the invoice.                      */
CREATE PROCEDURE sp_check_invoice_item(IN p_invoice INT UNSIGNED, IN p_dep INT UNSIGNED, IN p_plc INT UNSIGNED)
BEGIN
  DECLARE v_inv_so   INT UNSIGNED;
  DECLARE v_inv_stat VARCHAR(20);
  DECLARE v_item_so  INT UNSIGNED;

  SELECT service_order_id, status INTO v_inv_so, v_inv_stat FROM invoices WHERE invoice_id = p_invoice;
  IF v_inv_stat <> 'DRAFT' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoice_items: only DRAFT invoices can be changed';
  END IF;

  IF p_dep IS NOT NULL THEN
    SELECT jr.service_order_id INTO v_item_so
      FROM deployments d
      JOIN selections s ON s.selection_id = d.selection_id
      JOIN applications ap ON ap.application_id = s.application_id
      JOIN job_requirements jr ON jr.requirement_id = ap.requirement_id
     WHERE d.deployment_id = p_dep;
  ELSEIF p_plc IS NOT NULL THEN
    SELECT jr.service_order_id INTO v_item_so
      FROM placements p
      JOIN selections s ON s.selection_id = p.selection_id
      JOIN applications ap ON ap.application_id = s.application_id
      JOIN job_requirements jr ON jr.requirement_id = ap.requirement_id
     WHERE p.placement_id = p_plc;
  END IF;

  IF v_item_so IS NOT NULL AND v_item_so <> v_inv_so THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoice_items: billed deployment/placement belongs to a different service order';
  END IF;
END$$

CREATE TRIGGER trg_invoice_items_bi BEFORE INSERT ON invoice_items FOR EACH ROW
BEGIN CALL sp_check_invoice_item(NEW.invoice_id, NEW.deployment_id, NEW.placement_id); END$$
CREATE TRIGGER trg_invoice_items_bu BEFORE UPDATE ON invoice_items FOR EACH ROW
BEGIN CALL sp_check_invoice_item(NEW.invoice_id, NEW.deployment_id, NEW.placement_id); END$$

/* c) Payment allocation: same customer, invoice ISSUED, never over-allocate a
      payment or an invoice. Rows are locked (FOR UPDATE) to serialise races.
      p_old_payment/p_old_invoice identify the row being updated (excluded from sums). */
CREATE PROCEDURE sp_check_payment_allocation(IN p_payment INT UNSIGNED, IN p_invoice INT UNSIGNED,
                                             IN p_amount DECIMAL(14,2),
                                             IN p_old_payment INT UNSIGNED, IN p_old_invoice INT UNSIGNED)
BEGIN
  DECLARE v_pay_amt   DECIMAL(14,2);
  DECLARE v_pay_cust  INT UNSIGNED;
  DECLARE v_inv_total DECIMAL(14,2);
  DECLARE v_inv_stat  VARCHAR(20);
  DECLARE v_inv_cust  INT UNSIGNED;
  DECLARE v_pay_used  DECIMAL(14,2);
  DECLARE v_inv_used  DECIMAL(14,2);

  SELECT amount, customer_id INTO v_pay_amt, v_pay_cust FROM payments WHERE payment_id = p_payment FOR UPDATE;
  SELECT i.total_amount, i.status, ag.customer_id INTO v_inv_total, v_inv_stat, v_inv_cust
    FROM invoices i
    JOIN service_orders so ON so.service_order_id = i.service_order_id
    JOIN agreements ag     ON ag.agreement_id = so.agreement_id
   WHERE i.invoice_id = p_invoice FOR UPDATE OF i;

  IF v_pay_cust <> v_inv_cust THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'payment_allocations: payment and invoice belong to different customers';
  END IF;
  IF v_inv_stat <> 'ISSUED' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'payment_allocations: only ISSUED invoices can receive payments';
  END IF;

  SELECT COALESCE(SUM(allocated_amount), 0) INTO v_pay_used FROM payment_allocations
   WHERE payment_id = p_payment AND NOT (payment_id <=> p_old_payment AND invoice_id <=> p_old_invoice);
  SELECT COALESCE(SUM(allocated_amount), 0) INTO v_inv_used FROM payment_allocations
   WHERE invoice_id = p_invoice AND NOT (payment_id <=> p_old_payment AND invoice_id <=> p_old_invoice);

  IF v_pay_used + p_amount > v_pay_amt THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'payment_allocations: allocation exceeds the payment amount';
  END IF;
  IF v_inv_used + p_amount > v_inv_total THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'payment_allocations: allocation exceeds the invoice total';
  END IF;
END$$

CREATE TRIGGER trg_payment_allocations_bi BEFORE INSERT ON payment_allocations FOR EACH ROW
BEGIN CALL sp_check_payment_allocation(NEW.payment_id, NEW.invoice_id, NEW.allocated_amount, NULL, NULL); END$$
CREATE TRIGGER trg_payment_allocations_bu BEFORE UPDATE ON payment_allocations FOR EACH ROW
BEGIN CALL sp_check_payment_allocation(NEW.payment_id, NEW.invoice_id, NEW.allocated_amount, OLD.payment_id, OLD.invoice_id); END$$

/* d) Deployments: only DTS-payroll service orders create deployments; the employee
      must be the selected candidate; the supervisor must be a contact of the same customer. */
CREATE PROCEDURE sp_check_deployment(IN p_employee INT UNSIGNED, IN p_selection INT UNSIGNED, IN p_supervisor INT UNSIGNED)
BEGIN
  DECLARE v_model     VARCHAR(20);
  DECLARE v_cust      INT UNSIGNED;
  DECLARE v_sel_cand  INT UNSIGNED;
  DECLARE v_emp_cand  INT UNSIGNED;
  DECLARE v_sup_cust  INT UNSIGNED;

  SELECT st.employment_model, ag.customer_id, ap.candidate_id INTO v_model, v_cust, v_sel_cand
    FROM selections s
    JOIN applications ap     ON ap.application_id = s.application_id
    JOIN job_requirements jr ON jr.requirement_id = ap.requirement_id
    JOIN service_orders so   ON so.service_order_id = jr.service_order_id
    JOIN service_types st    ON st.service_type_id = so.service_type_id
    JOIN agreements ag       ON ag.agreement_id = so.agreement_id
   WHERE s.selection_id = p_selection;
  SELECT candidate_id INTO v_emp_cand FROM employees WHERE employee_id = p_employee;

  IF v_model <> 'DTS_PAYROLL' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'deployments: only DTS_PAYROLL service orders create deployments';
  END IF;
  IF v_emp_cand <> v_sel_cand THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'deployments: employee is not the candidate of this selection';
  END IF;
  IF p_supervisor IS NOT NULL THEN
    SELECT customer_id INTO v_sup_cust FROM customer_contacts WHERE contact_id = p_supervisor;
    IF v_sup_cust <> v_cust THEN
      SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'deployments: supervisor is not a contact of the customer';
    END IF;
  END IF;
END$$

CREATE TRIGGER trg_deployments_bi BEFORE INSERT ON deployments FOR EACH ROW
BEGIN CALL sp_check_deployment(NEW.employee_id, NEW.selection_id, NEW.customer_supervisor_contact_id); END$$
CREATE TRIGGER trg_deployments_bu BEFORE UPDATE ON deployments FOR EACH ROW
BEGIN CALL sp_check_deployment(NEW.employee_id, NEW.selection_id, NEW.customer_supervisor_contact_id); END$$

/* e) Invoice lifecycle: DRAFT -> ISSUED -> CANCELLED (final).
      - issuing requires header totals = sum of line items (and at least one line)
      - ISSUED / CANCELLED invoices are immutable (amounts, number, service order)
      - an invoice with payment allocations cannot be cancelled
      - lines can only be deleted while the invoice is DRAFT                      */
CREATE TRIGGER trg_invoices_bu BEFORE UPDATE ON invoices FOR EACH ROW
BEGIN
  DECLARE v_cnt INT;
  DECLARE v_sub DECIMAL(14,2);
  DECLARE v_tax DECIMAL(14,2);

  IF OLD.status <> 'DRAFT' AND (NEW.invoice_number <> OLD.invoice_number OR NEW.service_order_id <> OLD.service_order_id
       OR NEW.subtotal <> OLD.subtotal OR NEW.discount_amount <> OLD.discount_amount
       OR NEW.tax_total <> OLD.tax_total OR NEW.total_amount <> OLD.total_amount) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoices: an ISSUED or CANCELLED invoice cannot be modified';
  END IF;
  IF (OLD.status = 'CANCELLED' AND NEW.status <> 'CANCELLED') OR (OLD.status = 'ISSUED' AND NEW.status = 'DRAFT') THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoices: invalid status transition';
  END IF;
  IF OLD.status = 'DRAFT' AND NEW.status = 'ISSUED' THEN
    SELECT COUNT(*), COALESCE(SUM(taxable_value), 0), COALESCE(SUM(tax_amount), 0) INTO v_cnt, v_sub, v_tax
      FROM invoice_items WHERE invoice_id = OLD.invoice_id;
    IF v_cnt = 0 OR v_sub <> NEW.subtotal OR v_tax <> NEW.tax_total THEN
      SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoices: header subtotal/tax must equal the sum of the line items';
    END IF;
  END IF;
  IF NEW.status = 'CANCELLED' AND OLD.status <> 'CANCELLED'
     AND EXISTS (SELECT 1 FROM payment_allocations WHERE invoice_id = OLD.invoice_id) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoices: cannot cancel an invoice that has payment allocations';
  END IF;
END$$

CREATE TRIGGER trg_invoice_items_bd BEFORE DELETE ON invoice_items FOR EACH ROW
BEGIN
  IF (SELECT status FROM invoices WHERE invoice_id = OLD.invoice_id) <> 'DRAFT' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'invoice_items: only DRAFT invoices can be changed';
  END IF;
END$$

DELIMITER ;

/* ========================================================================== 
   VIEW: derived payment state (replaces storing PAID / PARTIALLY_PAID / OVERDUE)
   Only CLEARED payments count as received.
   ========================================================================== */
CREATE OR REPLACE VIEW v_invoice_balance AS
    SELECT 
        i.invoice_id,
        i.invoice_number,
        ag.customer_id,
        i.service_order_id,
        i.invoice_date,
        i.due_date,
        i.total_amount,
        COALESCE(SUM(CASE
                    WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount
                END),
                0) AS paid_amount,
        i.total_amount - COALESCE(SUM(CASE
                    WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount
                END),
                0) AS balance_amount,
        CASE
            WHEN i.status = 'CANCELLED' THEN 'CANCELLED'
            WHEN i.status = 'DRAFT' THEN 'DRAFT'
            WHEN
                COALESCE(SUM(CASE
                            WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount
                        END),
                        0) >= i.total_amount
            THEN
                'PAID'
            WHEN i.due_date < CURRENT_DATE() THEN 'OVERDUE'
            WHEN
                COALESCE(SUM(CASE
                            WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount
                        END),
                        0) > 0
            THEN
                'PARTIALLY_PAID'
            ELSE 'UNPAID'
        END AS payment_status
    FROM
        invoices i
            JOIN
        service_orders so ON so.service_order_id = i.service_order_id
            JOIN
        agreements ag ON ag.agreement_id = so.agreement_id
            LEFT JOIN
        payment_allocations pa ON pa.invoice_id = i.invoice_id
            LEFT JOIN
        payments p ON p.payment_id = pa.payment_id
            AND p.status = 'CLEARED'
    GROUP BY i.invoice_id , i.invoice_number , ag.customer_id , i.service_order_id , i.invoice_date , i.due_date , i.total_amount , i.status;

/* ===========================================================================
   PART C (optional) - fix v_invoice_balance
   For CANCELLED and DRAFT invoices nothing is receivable, but the original view still
   returned balance_amount = total_amount. Same view, balance now 0 for those two states.
   =========================================================================== */
CREATE OR REPLACE VIEW v_invoice_balance AS
    SELECT
        i.invoice_id, i.invoice_number, ag.customer_id, i.service_order_id, i.invoice_date, i.due_date, i.total_amount,
        COALESCE(SUM(CASE WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount END), 0) AS paid_amount,
        CASE WHEN i.status IN ('CANCELLED', 'DRAFT') THEN 0
             ELSE i.total_amount - COALESCE(SUM(CASE WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount END), 0)
        END AS balance_amount,
        CASE
            WHEN i.status = 'CANCELLED' THEN 'CANCELLED'
            WHEN i.status = 'DRAFT' THEN 'DRAFT'
            WHEN COALESCE(SUM(CASE WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount END), 0) >= i.total_amount THEN 'PAID'
            WHEN i.due_date < CURRENT_DATE() THEN 'OVERDUE'
            WHEN COALESCE(SUM(CASE WHEN p.payment_id IS NOT NULL THEN pa.allocated_amount END), 0) > 0 THEN 'PARTIALLY_PAID'
            ELSE 'UNPAID'
        END AS payment_status
    FROM invoices i
        JOIN service_orders so ON so.service_order_id = i.service_order_id
        JOIN agreements ag ON ag.agreement_id = so.agreement_id
        LEFT JOIN payment_allocations pa ON pa.invoice_id = i.invoice_id
        LEFT JOIN payments p ON p.payment_id = pa.payment_id AND p.status = 'CLEARED'
    GROUP BY i.invoice_id, i.invoice_number, ag.customer_id, i.service_order_id, i.invoice_date, i.due_date, i.total_amount, i.status;
